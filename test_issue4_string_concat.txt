// Test Issue 4: String concatenation in struct methods
// Testing if complex string concatenation with struct fields works

struct Player {
    var score:int;
    var lives:int;
    
    func printStatus() -> void {
        // This should work but might cause errors
        print("Score: " + score + " Lives: " + lives);
    }
    
    func printStatusWorkaround() -> void {
        // Workaround using separate print statements
        print("Score: ");
        print(score);
        print(" Lives: ");
        print(lives);
        print("\n");
    }
}

func main() -> void {
    var player:Player;
    player.score = 100;
    player.lives = 3;
    
    // Try the problematic version
    player.printStatus();
}
