// Design for heap-allocated mutable strings

// Approach 1: malloc/free built-ins
func example1() -> void {
    // Allocate 100 bytes for a mutable string
    var buffer:string = malloc(100);
    
    // Now we can modify it
    buffer[0] = 'H';
    buffer[1] = 'i';
    buffer[2] = '\0';  // null terminator
    
    print("Buffer: " + buffer);
    
    // Use strcpy to copy a string literal
    strcpy(buffer, "Hello World");
    print("After strcpy: " + buffer);
    
    // Modify individual characters
    buffer[0] = 'J';
    print("After modification: " + buffer);
    
    // Free when done
    free(buffer);
}

// Approach 2: strdup built-in (duplicates a string on heap)
func example2() -> void {
    // Create a mutable copy of a string literal
    var mutable:string = strdup("Hello");
    
    // Now we can modify it
    mutable[0] = 'J';
    print("Modified: " + mutable);
    
    // Free when done
    free(mutable);
}

// Approach 3: Combined with string functions
func example3() -> void {
    var str1:string = strdup("Hello");
    var str2:string = strdup(" World");
    
    // Concatenate
    strcat(str1, str2);  // Dangerous if str1 buffer is too small!
    
    print("Result: " + str1);
    
    free(str1);
    free(str2);
}
