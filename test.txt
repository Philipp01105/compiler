func factorial(n:int) -> int
{
    var result:int = 1;
    for (var i:int = 1; i <= n; i++) {
        var temp:int = result * i;
        result = temp;
    }
    return result;
}

func fibonacci(n:int) -> int
{
    if (n <= 1) {
        return n;
    }

    var prev:int = 0;
    var curr:int = 1;

    for (var i:int = 2; i <= n; i++) {
        var next:int = prev + curr;
        prev = curr;
        curr = next;
    }

    return curr;
}

func power(base:int, exp:int) -> int
{
    var result:int = 1;
    for (var i:int = 0; i < exp; i++) {
        var temp:int = result * base;
        result = temp;
    }
    return result;
}

func sumOfSquares(n:int) -> int
{
    var sum:int = 0;
    for (var i:int = 1; i <= n; i++) {
        var sq:int = i * i;
        var temp:int = sum + sq;

func gcd(a:int, b:int) -> int
{
    for (var i:int = 0; i < 1000; i++) {
        if (b == 0) {
            return a;
        }
        var temp:int = b;
        b = a % b;
        a = temp;
    }
    return a;
}

func isPrime(n:int) -> bit
{
    if (n <= 1) {
        return 0;
    }
    if (n == 2) {
        return 1;
    }
    if (n % 2 == 0) {
        return 0;
    }

    var i:int = 3;
    for (var count:int = 0; count < 1000; count++) {
        if (i * i > n) {
            return 1;
        }
        if (n % i == 0) {
            return 0;
        }
        i = i + 2;
    }

    return 1;
}

        height = h;

        if (width == height) {

            return 0;
        }
    }
}
