// Test 8: Fakultät (iterativ)
func factorial(n:int) -> int
{
    var result = 1;
    for(var i = 1; i <= n; i++)
    {
        result *= i;
    }
    return result;
}

// Test 9: Fibonacci
func fibonacci(n:int) -> int
{
    var a = 0;
    var b = 1;
    
    for(var i = 0; i < n; i++)
    {
        var temp = a + b;
        a = b;
        b = temp;
    }
    
    return a;
}

// Test 10: Potenz
func power(base:int, exp:int) -> int
{
    var result = 1;
    for(var i = 0; i < exp; i++)
    {
        result *= base;
    }
    return result;
}

// Test 11: Maximum von zwei Zahlen
func max(a:int, b:int) -> int
{
    var result = a;
    var diff = b - a;
    
    // Primitive if-Simulation
    for(var i = 0; i < 1; i++)
    {
        // Wenn diff > 0, dann b > a
        var isGreater = 0;
        for(var j = 0; j < diff; j++)
        {
            isGreater = 1;
        }
        
        // Setze result = b wenn isGreater
        var temp = b * isGreater;
        var temp2 = result * (1 - isGreater);
        result = temp + temp2;
    }
    
    return result;
}

// Test 12: Modulo-Berechnung (n mod d)
func modulo(n:int, d:int) -> int
{
    var quotient = n / d;
    var product = quotient * d;
    var remainder = n - product;
    return remainder;
}

// Test 13: Ist Gerade?
func isEven(n:int) -> int
{
    var mod = modulo(n, 2);

    var result = 1 - mod;
    
    return result;
}

// Test 14: GGT (Größter Gemeinsamer Teiler)
func gcd(a:int, b:int) -> int
{
    // Euklid-Algorithmus: wiederhole bis b == 0
    for(var i = 0; i < 100; i++)
    {
        // Prüfe ob b == 0
        var bIsZero = 1;
        for(var j = 0; j < b; j++)
        {
            bIsZero = 0;
        }

        // Wenn b == 0, sind wir fertig, mache nichts mehr
        // Ansonsten: berechne neues a und b

        // Berechne "shouldContinue" = 1 wenn b != 0
        var shouldContinue = 0;
        for(var check = 0; check < b; check++)
        {
            shouldContinue = 1;
        }

        // Nur wenn shouldContinue == 1, dann rechne weiter
        // (sonst würden wir durch 0 teilen!)
        for(var do_it = 0; do_it < shouldContinue; do_it++)
        {
            var temp = b;
            b = modulo(a, b);
            a = temp;
        }
    }
    
    return a;
}
