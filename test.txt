// Test 11: Fakultaet (iterativ)
func factorial(n:int) -> int
{
    var result = 1;
    for(var i = 1; i <= n; i++)
    {
        result *= i;
    }
    return result;
}

// Test 12: Fibonacci
func fibonacci(n:int) -> int
{
    var a = 0;
    var b = 1;

    for(var i = 0; i < n; i++)
    {
        var temp = a + b;
        a = b;
        b = temp;
    }

    return a;
}

// Test 13: Potenz
func power(base:int, exp:int) -> int
{
    var result = 1;
    for(var i = 0; i < exp; i++)
    {
        result *= base;
    }
    return result;
}

// Test 14: Summe von 1 bis n
func sumTo(n:int) -> int
{
    var sum = 0;
    for(var i = 1; i <= n; i++)
    {
        sum += i;
    }
    return sum;
}

// Test 15: Modulo-Berechnung
func modulo(n:int, d:int) -> int
{
    var quotient = n / d;
    var product = quotient * d;
    var remainder = n - product;
    return remainder;
}

// Test 16: Ist Gerade?
func isEven(n:int) -> int
{
    var mod = modulo(n, 2);
    var result = 1 - mod;
    return result;
}

// Test 17: Absolute Zahl (ohne if)
func abs(n:int) -> int
{
    var isNegative = 0;
    var temp = 0 - n;

    for(var i = 0; i < temp; i++)
    {
        isNegative = 1;
    }

    var positive = n;
    var negative = 0 - n;

    var result = positive;
    for(var j = 0; j < isNegative; j++)
    {
        result = negative;
    }

    return result;
}

// Test 18: Maximum von zwei Zahlen (ohne if)
func max(a:int, b:int) -> int
{
    var result = a;
    var diff = b - a;

    for(var i = 0; i < 1; i++)
    {
        var isGreater = 0;
        for(var j = 0; j < diff; j++)
        {
            isGreater = 1;
        }

        var temp = b * isGreater;
        var temp2 = result * (1 - isGreater);
        result = temp + temp2;
    }

    return result;
}

// Test 19: Minimum von zwei Zahlen
func min(a:int, b:int) -> int
{
    var maxVal = max(a, b);
    var sum = a + b;
    return sum - maxVal;
}

// Test 20: GGT (Groesster Gemeinsamer Teiler)
func gcd(a:int, b:int) -> int
{
    for(var i = 0; i < 100; i++)
    {
        var shouldContinue = 0;
        for(var check = 0; check < b; check++)
        {
            shouldContinue = 1;
        }

        for(var do_it = 0; do_it < shouldContinue; do_it++)
        {
            var temp = b;
            b = modulo(a, b);
            a = temp;
        }
    }

    return a;
}

// Test 21: Verschachtelte Aufrufe - Level 1
func addAndMultiply(a:int, b:int, c:int) -> int
{
    var sum = add(a, b);
    return multiply(sum, c);
}

// Test 22: Verschachtelte Aufrufe - Level 2
func complexCalc(a:int, b:int, c:int, d:int) -> int
{
    var part1 = multiply(add(a, b), c);
    var part2 = divide(subtract(d, a), b);
    return add(part1, part2);
}

// Test 23: Summe der Quadrate
func sumOfSquares(a:int, b:int) -> int
{
    var sq1 = square(a);
    var sq2 = square(b);
    return add(sq1, sq2);
}

// Test 24: Durchschnitt von drei Zahlen
func average3(a:int, b:int, c:int) -> int
{
    var sum = sum3(a, b, c);
    return divide(sum, 3);
}
