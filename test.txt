// Test 16: Fakultaet (iterativ)
func factorial(n:int) -> int
{
    var result = 1;
    for(var i = 1; i <= n; i++)
    {
        result *= i;
    }
    return result;
}

// Test 17: Fibonacci
func fibonacci(n:int) -> int
{
    var a = 0;
    var b = 1;

    for(var i = 0; i < n; i++)
    {
        var temp = a + b;
        a = b;
        b = temp;
    }

    return a;
}

// Test 18: Potenz
func power(base:int, exp:int) -> int
{
    var result = 1;
    for(var i = 0; i < exp; i++)
    {
        result *= base;
    }
    return result;
}

// Test 19: Summe von 1 bis n
func sumTo(n:int) -> int
{
    var sum = 0;
    for(var i = 1; i <= n; i++)
    {
        sum += i;
    }
    return sum;
}

// Test 20: Modulo-Berechnung
func modulo(n:int, d:int) -> int
{
    var quotient = n / d;
    var product = quotient * d;
    var remainder = n - product;
    return remainder;
}

// Test 21: Ist Gerade?
func isEven(n:int) -> bit
{
    var mod = modulo(n, 2);
    if (mod == 0) {
        return 1;
    } else {
        return 0;
    }
}

// Test 23: Minimum mit if/else
func minWithIf(a:int, b:int) -> int

// Test 24: GGT (Groesster Gemeinsamer Teiler) mit if/else
func gcdWithIf(a:int, b:int) -> int
{
    for(var i = 0; i < 100; i++)
    {
        if (b > 0) {
            var temp = b;
            b = modulo(a, b);
            a = temp;
        }
    }
    return a;
}

// Test 25: Verschachtelte Aufrufe - Level 1
func addAndMultiply(a:int, b:int, c:int) -> int
{
    var sum = add(a, b);
    return multiply(sum, c);
}

// Test 26: Verschachtelte Aufrufe - Level 2
func complexCalc(a:int, b:int, c:int, d:int) -> int
{
    var part1 = multiply(add(a, b), c);
    var part2 = divide(subtract(d, a), b);
    return add(part1, part2);
}

// Test 27: Summe der Quadrate
func sumOfSquares(a:int, b:int) -> int
{
    var sq1 = square(a);
    var sq2 = square(b);
    return add(sq1, sq2);
}

// Test 28: Durchschnitt von drei Zahlen
func average3(a:int, b:int, c:int) -> int
{
    var sum = sum3(a, b, c);
    return divide(sum, 3);
}
