func printLabeled(label:string, value:int) -> void {
    print(label);
    print(": ");
    print(value);
}

func arrayFillInt(arr:int[], size:int, value:int) -> void {
    for (var i:int = 0; i < size; i++) {
        arr[i] = value;
    }
}

func main() -> void {
    print("Before array init");
    var[10] intArr:int;
    print("After array declaration");
    arrayFillInt(intArr, 10, 42);
    print("After arrayFillInt");
    printLabeled("intArr[0]", intArr[0]);
    printLabeled("intArr[5]", intArr[5]);
    print("Done");
}
