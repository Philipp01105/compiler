func printLabeledBool(label:string, value:bit) -> void {
    print(label);
    print(": ");
    print(value);
}

func boolAnd(a:bit, b:bit) -> bit {
    if (a == 1 && b == 1) {
        return 1;
    }
    return 0;
}

func main() -> void {
    printLabeledBool("boolAnd(1, 1)", boolAnd(1, 1));
    printLabeledBool("boolAnd(1, 0)", boolAnd(1, 0));
}
