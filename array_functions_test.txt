// Print helper functions
func printSeparator() -> void

// Fill array with a specific value
func fillArray(arr[]:int, size:int, value:int) -> void
{
    for (var i:int = 0; i < size; i++) {
        arr[i] = value;
    }
}

// Print array elements
func printArray(arr[]:int, size:int, name:string) -> void
{
    print("  Array " + name + ":");
    for (var i:int = 0; i < size; i++) {
        print("    [" + i + "] = " + arr[i]);
    }
}

// Sum all elements in array
func sumArray(arr[]:int, size:int) -> int
{
    var sum:int = 0;
    for (var i:int = 0; i < size; i++) {
        var temp:int = sum + arr[i];
        sum = temp;
    }
    return sum;
}

// Find maximum element in array
func maxArray(arr[]:int, size:int) -> int
{
    if (size == 0) {
        return 0;
    }
    var max:int = arr[0];
    for (var i:int = 1; i < size; i++) {
        if (arr[i] > max) {
            max = arr[i];
        }
    }
    return max;
}

// Find minimum element in array
func minArray(arr[]:int, size:int) -> int
{
    if (size == 0) {
        return 0;
    }
    var min:int = arr[0];
    for (var i:int = 1; i < size; i++) {
        if (arr[i] < min) {
            min = arr[i];
        }
    }
    return min;
}

// Copy array from source to destination
func copyArray(src[]:int, dst[]:int, size:int) -> void
{
    for (var i:int = 0; i < size; i++) {
        dst[i] = src[i];
    }
}

// Reverse array in place
func reverseArray(arr[]:int, size:int) -> void
{
    for (var i:int = 0; i < size / 2; i++) {
        var j:int = size - 1 - i;
        var temp:int = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }
}

// Count occurrences of a value
func countValue(arr[]:int, size:int, value:int) -> int
{
    var count:int = 0;
    for (var i:int = 0; i < size; i++) {
        if (arr[i] == value) {
            var temp:int = count + 1;
            count = temp;
        }
    }
    return count;
}
