// Comprehensive test suite for struct with methods feature
// Tests all functionality systematically

// Test 1: Struct with different field types
struct Person {
    int age;
    char initial;
    float height;
    bit isActive;
    
    func setAge(a:int) -> void {
        age = a;
    }
    
    func getAge() -> int {
        return age;
    }
    
    func setInitial(i:char) -> void {
        initial = i;
    }
    
    func getInitial() -> char {
        return initial;
    }
    
    func setHeight(h:float) -> void {
        height = h;
    }
    
    func getHeight() -> float {
        return height;
    }
    
    func activate() -> void {
        isActive = 1;
    }
    
    func deactivate() -> void {
        isActive = 0;
    }
    
    func isPersonActive() -> bit {
        return isActive;
    }
    
    func incrementAge() -> void {
        age = age + 1;
    }
}

// Test 2: Struct with calculations
struct Calculator {
    int result;
    
    func add(a:int, b:int) -> int {
        result = a + b;
        return result;
    }
    
    func subtract(a:int, b:int) -> int {
        result = a - b;
        return result;
    }
    
    func multiply(a:int, b:int) -> int {
        result = a * b;
        return result;
    }
    
    func divide(a:int, b:int) -> int {
        result = a / b;
        return result;
    }
    
    func getResult() -> int {
        return result;
    }
    
    func clear() -> void {
        result = 0;
    }
}

// Test 3: Struct with conditional logic
struct Validator {
    int value;
    int min;
    int max;
    
    func setValue(v:int) -> void {
        value = v;
    }
    
    func setRange(minVal:int, maxVal:int) -> void {
        min = minVal;
        max = maxVal;
    }
    
    func isValid() -> bit {
        if (value >= min && value <= max) {
            return 1;
        } else {
            return 0;
        }
    }
    
    func isEven() -> bit {
        if (value % 2 == 0) {
            return 1;
        } else {
            return 0;
        }
    }
    
    func isPositive() -> bit {
        if (value > 0) {
            return 1;
        } else {
            return 0;
        }
    }
}

func main() -> void {
    print("=================================================================");
    print("          COMPREHENSIVE STRUCT TEST SUITE");
    print("=================================================================");
    print("");
    
    // Test 1: Person struct - Testing all data types
    print("=== TEST 1: Person Struct (Multiple Field Types) ===");
    print("");
    
    var person:Person;
    
    // Test setAge and getAge
    person.setAge(25);
    print("Test 1.1 - setAge(25), getAge():");
    print("  Age: " + person.getAge());
    print("  Expected: 25");
    
    // Test incrementAge
    person.incrementAge();
    print("Test 1.2 - incrementAge():");
    print("  Age: " + person.getAge());
    print("  Expected: 26");
    
    // Test setInitial and getInitial
    person.setInitial('J');
    print("Test 1.3 - setInitial('J'), getInitial():");
    print("  Initial: " + person.getInitial());
    print("  Expected: J");
    
    // Test activate and isPersonActive
    person.activate();
    print("Test 1.4 - activate():");
    if (person.isPersonActive() == 1) {
        print("  Status: Active");
        print("  Expected: Active");
    }
    
    // Test deactivate
    person.deactivate();
    print("Test 1.5 - deactivate():");
    if (person.isPersonActive() == 0) {
        print("  Status: Inactive");
        print("  Expected: Inactive");
    }
    
    // Test direct field access
    person.age = 30;
    person.initial = 'M';
    print("Test 1.6 - Direct field access:");
    print("  Age: " + person.age);
    print("  Initial: " + person.initial);
    print("  Expected: Age=30, Initial=M");
    
    print("");
    
    // Test 2: Calculator struct - Testing arithmetic methods
    print("=== TEST 2: Calculator Struct (Arithmetic Operations) ===");
    print("");
    
    var calc:Calculator;
    
    // Test add
    print("Test 2.1 - add(10, 5):");
    print("  Result: " + calc.add(10, 5));
    print("  Expected: 15");
    
    // Test subtract
    print("Test 2.2 - subtract(20, 8):");
    print("  Result: " + calc.subtract(20, 8));
    print("  Expected: 12");
    
    // Test multiply
    print("Test 2.3 - multiply(7, 6):");
    print("  Result: " + calc.multiply(7, 6));
    print("  Expected: 42");
    
    // Test divide
    print("Test 2.4 - divide(100, 4):");
    print("  Result: " + calc.divide(100, 4));
    print("  Expected: 25");
    
    // Test getResult
    calc.multiply(3, 4);
    print("Test 2.5 - multiply(3, 4), getResult():");
    print("  Result: " + calc.getResult());
    print("  Expected: 12");
    
    // Test clear
    calc.clear();
    print("Test 2.6 - clear():");
    print("  Result: " + calc.getResult());
    print("  Expected: 0");
    
    // Test direct field access
    calc.result = 99;
    print("Test 2.7 - Direct field access (result = 99):");
    print("  Result: " + calc.result);
    print("  Expected: 99");
    
    print("");
    
    // Test 3: Validator struct - Testing conditional methods
    print("=== TEST 3: Validator Struct (Conditional Logic) ===");
    print("");
    
    var validator:Validator;
    
    // Set up range
    validator.setRange(0, 100);
    
    // Test isValid - valid case
    validator.setValue(50);
    print("Test 3.1 - setValue(50), setRange(0,100), isValid():");
    if (validator.isValid() == 1) {
        print("  Status: Valid");
        print("  Expected: Valid");
    }
    
    // Test isValid - invalid case
    validator.setValue(150);
    print("Test 3.2 - setValue(150), isValid():");
    if (validator.isValid() == 0) {
        print("  Status: Invalid");
        print("  Expected: Invalid");
    }
    
    // Test isEven - even number
    validator.setValue(10);
    print("Test 3.3 - setValue(10), isEven():");
    if (validator.isEven() == 1) {
        print("  Status: Even");
        print("  Expected: Even");
    }
    
    // Test isEven - odd number
    validator.setValue(7);
    print("Test 3.4 - setValue(7), isEven():");
    if (validator.isEven() == 0) {
        print("  Status: Odd");
        print("  Expected: Odd");
    }
    
    // Test isPositive - positive
    validator.setValue(42);
    print("Test 3.5 - setValue(42), isPositive():");
    if (validator.isPositive() == 1) {
        print("  Status: Positive");
        print("  Expected: Positive");
    }
    
    // Test isPositive - negative
    validator.setValue(-5);
    print("Test 3.6 - setValue(-5), isPositive():");
    if (validator.isPositive() == 0) {
        print("  Status: Not Positive");
        print("  Expected: Not Positive");
    }
    
    // Test direct field access
    validator.value = 75;
    validator.min = 50;

    print("Test 3.7 - Direct field access:");
    print("  Value: " + validator.value);
    print("  Min: " + validator.min);

    }
    
    print("");
    
    // Test 4: Multiple instances
    print("=== TEST 4: Multiple Struct Instances ===");
    print("");

    print("  Person1 Age: " + person1.getAge());

        print("  Person1: Active");

        print("  Person2: Inactive");
    }
    print("  Expected: Person1=Active, Person2=Inactive");
    
    print("");
    
    // Test 5: Chained method calls (via intermediate storage)
    print("=== TEST 5: Method Return Values ===");
    print("");
    
    var calc2:Calculator;
    var sum:int = calc2.add(5, 3);
    var prod:int = calc2.multiply(sum, 2);
    
    print("Test 5.1 - Using return values:");
    print("  sum = add(5, 3) = " + sum);
    print("  prod = multiply(sum, 2) = " + prod);
    print("  Expected: sum=8, prod=16");
    
    print("");
    print("=================================================================");
    print("          ALL TESTS COMPLETED");
    print("=================================================================");
