// Unit tests for struct feature - Testing specific scenarios

// Test struct with single field and single method
struct Simple {
    int value;
    
    func getValue() -> int {
        return value;
    }
}

// Test struct with no methods (fields only)
struct FieldsOnly {
    int x;
    int y;
    int z;
}

// Test struct with methods that modify multiple fields
struct Point3D {
    int x;
    int y;
    int z;
    
    func setPosition(px:int, py:int, pz:int) -> void {
        x = px;
        y = py;
        z = pz;
    }
    
    func reset() -> void {
        x = 0;
        y = 0;
        z = 0;
    }
    
    func getX() -> int {
        return x;
    }
    
    func getY() -> int {
        return y;
    }
    
    func getZ() -> int {
        return z;
    }
    
    func sumCoordinates() -> int {
        return x + y + z;
    }
}

// Test struct with complex conditionals
struct Range {
    int start;
    int end;
    
    func setRange(s:int, e:int) -> void {
        start = s;
        end = e;
    }
    
    func contains(value:int) -> bit {
        if (value >= start && value <= end) {
            return 1;
        } else {
            return 0;
        }
    }
    
    func getLength() -> int {
        return end - start;
    }
    
    func expand(amount:int) -> void {
        start = start - amount;
        end = end + amount;
    }
}

// Test struct with bit return values
struct Flags {
    bit flag1;
    bit flag2;
    bit flag3;
    
    func setAll() -> void {
        flag1 = 1;
        flag2 = 1;
        flag3 = 1;
    }
    
    func clearAll() -> void {
        flag1 = 0;
        flag2 = 0;
        flag3 = 0;
    }
    
    func toggle1() -> void {
        if (flag1 == 1) {
            flag1 = 0;
        } else {
            flag1 = 1;
        }
    }
    
    func allSet() -> bit {
        if (flag1 == 1 && flag2 == 1 && flag3 == 1) {
            return 1;
        } else {
            return 0;
        }
    }
    
    func anySet() -> bit {
        if (flag1 == 1 || flag2 == 1 || flag3 == 1) {
            return 1;
        } else {
            return 0;
        }
    }
    
    func noneSet() -> bit {
        if (flag1 == 0 && flag2 == 0 && flag3 == 0) {
            return 1;
        } else {
            return 0;
        }
    }
}

func main() -> void {
    print("=================================================================");
    print("          STRUCT UNIT TESTS");
    print("=================================================================");
    print("");
    
    // Unit Test 1: Simple struct
    print("--- Unit Test 1: Simple Struct ---");
    var s:Simple;
    s.value = 42;
    print("Set value to 42:");
    print("  getValue() = " + s.getValue());
    print("  Pass: " + (s.getValue() == 42));
    print("");
    
    // Unit Test 2: Fields only struct
    print("--- Unit Test 2: Fields Only Struct ---");
    var fo:FieldsOnly;
    fo.x = 1;
    fo.y = 2;
    fo.z = 3;
    print("Set x=1, y=2, z=3:");
    print("  x = " + fo.x);
    print("  y = " + fo.y);
    print("  z = " + fo.z);
    print("  Pass: " + (fo.x == 1 && fo.y == 2 && fo.z == 3));
    print("");
    
    // Unit Test 3: Point3D - setPosition
    print("--- Unit Test 3: Point3D - setPosition ---");
    var p:Point3D;
    p.setPosition(10, 20, 30);
    print("setPosition(10, 20, 30):");
    print("  x = " + p.getX());
    print("  y = " + p.getY());
    print("  z = " + p.getZ());
    print("  Pass: " + (p.getX() == 10 && p.getY() == 20 && p.getZ() == 30));
    print("");
    
    // Unit Test 4: Point3D - sumCoordinates
    print("--- Unit Test 4: Point3D - sumCoordinates ---");
    p.setPosition(5, 10, 15);
    var sum:int = p.sumCoordinates();
    print("setPosition(5, 10, 15), sumCoordinates():");
    print("  Sum = " + sum);
    print("  Pass: " + (sum == 30));
    print("");
    
    // Unit Test 5: Point3D - reset
    print("--- Unit Test 5: Point3D - reset ---");
    p.reset();
    print("reset():");
    print("  x = " + p.getX());
    print("  y = " + p.getY());
    print("  z = " + p.getZ());
    print("  Pass: " + (p.getX() == 0 && p.getY() == 0 && p.getZ() == 0));
    print("");
    
    // Unit Test 6: Range - contains
    print("--- Unit Test 6: Range - contains ---");
    var r:Range;
    r.setRange(10, 20);
    print("setRange(10, 20):");
    print("  contains(15) = " + r.contains(15));
    print("  contains(25) = " + r.contains(25));
    print("  contains(10) = " + r.contains(10));
    print("  contains(20) = " + r.contains(20));
    print("  Pass: contains works correctly");
    print("");
    
    // Unit Test 7: Range - getLength
    print("--- Unit Test 7: Range - getLength ---");
    r.setRange(5, 15);
    var len:int = r.getLength();
    print("setRange(5, 15), getLength():");
    print("  Length = " + len);
    print("  Pass: " + (len == 10));
    print("");
    
    // Unit Test 8: Range - expand
    print("--- Unit Test 8: Range - expand ---");
    r.setRange(10, 20);
    r.expand(5);
    print("setRange(10, 20), expand(5):");
    print("  start = " + r.start);
    print("  end = " + r.end);
    print("  Pass: " + (r.start == 5 && r.end == 25));
    print("");
    
    // Unit Test 9: Flags - setAll
    print("--- Unit Test 9: Flags - setAll ---");
    var f:Flags;
    f.setAll();
    print("setAll():");
    print("  allSet() = " + f.allSet());
    print("  Pass: " + (f.allSet() == 1));
    print("");
    
    // Unit Test 10: Flags - clearAll
    print("--- Unit Test 10: Flags - clearAll ---");
    f.clearAll();
    print("clearAll():");
    print("  noneSet() = " + f.noneSet());
    print("  Pass: " + (f.noneSet() == 1));
    print("");
    
    // Unit Test 11: Flags - toggle
    print("--- Unit Test 11: Flags - toggle ---");
    f.clearAll();
    f.toggle1();
    print("clearAll(), toggle1():");
    print("  flag1 = " + f.flag1);
    print("  anySet() = " + f.anySet());
    print("  Pass: " + (f.flag1 == 1 && f.anySet() == 1));
    print("");
    
    // Unit Test 12: Flags - anySet with mixed flags
    print("--- Unit Test 12: Flags - anySet with mixed flags ---");
    f.clearAll();
    f.flag2 = 1;
    print("clearAll(), set flag2=1:");
    print("  anySet() = " + f.anySet());
    print("  allSet() = " + f.allSet());
    print("  Pass: " + (f.anySet() == 1 && f.allSet() == 0));
    print("");
    
    // Unit Test 13: Multiple instances independence
    print("--- Unit Test 13: Multiple Instances Independence ---");
    var p1:Point3D;
    var p2:Point3D;
    p1.setPosition(1, 2, 3);
    p2.setPosition(10, 20, 30);
    print("p1.setPosition(1,2,3), p2.setPosition(10,20,30):");
    print("  p1.x = " + p1.x);
    print("  p2.x = " + p2.x);
    print("  Pass: " + (p1.x == 1 && p2.x == 10));
    print("");
    
    // Unit Test 14: Direct field modification vs method
    print("--- Unit Test 14: Direct Field vs Method ---");
    var s2:Simple;
    s2.value = 100;
    var v1:int = s2.getValue();
    s2.value = 200;
    var v2:int = s2.getValue();
    print("Direct field: value=100, getValue()=" + v1);
    print("Direct field: value=200, getValue()=" + v2);
    print("  Pass: " + (v1 == 100 && v2 == 200));
    print("");
    
    // Unit Test 15: Nested method calls
    print("--- Unit Test 15: Chained Operations ---");
    var r2:Range;
    r2.setRange(0, 10);
    var len1:int = r2.getLength();
    r2.expand(5);
    var len2:int = r2.getLength();
    print("setRange(0,10): length=" + len1);
    print("After expand(5): length=" + len2);
    print("  Pass: " + (len1 == 10 && len2 == 20));
    print("");
    
    print("=================================================================");
    print("          ALL UNIT TESTS COMPLETED");
    print("=================================================================");
}
