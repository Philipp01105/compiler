// Test file for struct with methods feature

struct Point {
    int x;
    int y;
    
    func move(dx:int, dy:int) -> void {
        x = x + dx;
        y = y + dy;
    }
    
    func getX() -> int {
        return x;
    }
    
    func getY() -> int {
        return y;
    }
}

func main() -> void {
    var p:Point;
    
    // Initialize fields
    p.x = 10;
    p.y = 20;
    
    print("Initial position:");
    print("  x = " + p.x);
    print("  y = " + p.y);
    
    // Call move method
    p.move(5, 4);
    
    print("After moving by (5, 4):");
    print("  x = " + p.getX());
    print("  y = " + p.getY());
}
