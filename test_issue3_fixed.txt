// Test for Issue 3: Arrays as struct fields (FIXED)
// This demonstrates that arrays can now be declared as struct fields

struct TicTacToe {
    int board[9];  // Array field - 3x3 board
    
    func setCell(index:int, value:int) -> void {
        board[index] = value;
    }
    
    func getCell(index:int) -> int {
        return board[index];
    }
    
    func initBoard() -> void {
        board[0] = 0;
        board[1] = 0;
        board[2] = 0;
        board[3] = 0;
        board[4] = 0;
        board[5] = 0;
        board[6] = 0;
        board[7] = 0;
        board[8] = 0;
    }
}

struct DataStore {
    int numbers[5];
    
    func sum() -> int {
        var total:int;
        total = numbers[0];
        total = total + numbers[1];
        total = total + numbers[2];
        total = total + numbers[3];
        total = total + numbers[4];
        return total;
    }
}

func main() -> void {
    // Test 1: TicTacToe board
    var game:TicTacToe;
    game.initBoard();
    game.setCell(0, 1);  // X in top-left
    game.setCell(4, 2);  // O in center
    game.setCell(8, 1);  // X in bottom-right
    
    var cell0:int;
    var cell4:int;
    var cell8:int;
    cell0 = game.getCell(0);  // Should be 1
    cell4 = game.getCell(4);  // Should be 2
    cell8 = game.getCell(8);  // Should be 1
    
    // Test 2: DataStore with sum
    var store:DataStore;
    store.numbers[0] = 10;
    store.numbers[1] = 20;
    store.numbers[2] = 30;
    store.numbers[3] = 40;
    store.numbers[4] = 50;
    
    var total:int;
    total = store.sum();  // Should be 150
}
