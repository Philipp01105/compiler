// Test Issue 6: For loop decrement syntax
// Testing if i++ and i-- work in for loops

func main() -> void {
    var sum:int;
    sum = 0;
    
    // Test 1: i++ (increment)
    for (var i:int = 0; i < 5; i++) {
        sum = sum + i;
    }
    // sum should be 0+1+2+3+4 = 10
    
    // Test 2: i-- (decrement)
    var countdown:int;
    countdown = 0;
    for (var j:int = 5; j > 0; j--) {
        countdown = countdown + j;
    }
    // countdown should be 5+4+3+2+1 = 15
    
    // Test 3: i = i + 1
    var test3:int;
    test3 = 0;
    for (var k:int = 0; k < 3; k = k + 1) {
        test3 = test3 + k;
    }
    // test3 should be 0+1+2 = 3
    
    // Test 4: i = i - 1
    var test4:int;
    test4 = 0;
    for (var m:int = 3; m > 0; m = m - 1) {
        test4 = test4 + m;
    }
    // test4 should be 3+2+1 = 6
}
