// =================================================================
// NUMBER GUESSING GAME
// A simple number guessing game simulation
// =================================================================
// This game demonstrates:
// - Struct usage with methods
// - Control flow (if-else chains)
// - Game logic and state management
// - Range checking and validation
// =================================================================

#import <stdlib/core.txt>
#import <stdlib/utils.txt>

// Number Guessing Game struct
struct GuessGame {
    int secretNumber;
    int guessCount;
    int maxGuesses;
    int gameWon;
    int rangeLow;
    int rangeHigh;
    
    // Initialize the game
    func init(secret:int, maxAttempts:int) -> void {
        secretNumber = secret;
        guessCount = 0;
        maxGuesses = maxAttempts;
        gameWon = 0;
        rangeLow = 1;
        rangeHigh = 100;
    }
    
    // Make a guess
    func makeGuess(guess:int) -> int {
        guessCount = guessCount + 1;
        
        if (guess == secretNumber) {
            gameWon = 1;
            return 0;
        } else if (guess < secretNumber) {
            if (guess > rangeLow) {
                rangeLow = guess;
            }
            return -1;
        } else {
            if (guess < rangeHigh) {
                rangeHigh = guess;
            }
            return 1;
        }
    }
    
    // Check if game is over
    func isGameOver() -> bit {
        if (gameWon == 1) {
            return 1;
        }
        if (guessCount >= maxGuesses) {
            return 1;
        }
        return 0;
    }
    
    // Print game status
    func printStatus() -> void {
        printLine();
        print("Guess Number Game");
        printLabeled("Range Low", rangeLow);
        printLabeled("Range High", rangeHigh);
        printLabeled("Guesses used", guessCount);
        printLabeled("Max guesses", maxGuesses);
        printLine();
    }
}

// Simulate a game
func simulateGame() -> void {
    printDoubleLine();
    print("         NUMBER GUESSING GAME");
    printDoubleLine();
    print("");
    
    var game:GuessGame;
    game.init(42, 7);
    
    print("Secret number is between 1 and 100");
    print("You have 7 guesses to find it!");
    print("");
    
    // Array of guesses to simulate
    var[7] guesses:int;
    guesses[0] = 50;
    guesses[1] = 25;
    guesses[2] = 37;
    guesses[3] = 43;
    guesses[4] = 40;
    guesses[5] = 41;
    guesses[6] = 42;
    
    for (var i:int = 0; i < 7; i++) {
        if (game.isGameOver() == 1) {
            break;
        }
        
        var guess:int = guesses[i];
        print("Guess #" + (i + 1) + ": " + guess);
        
        var result:int = game.makeGuess(guess);
        
        if (result == 0) {
            print("CORRECT! You found the number!");
            break;
        } else if (result < 0) {
            print("Too low! Try higher.");
        } else {
            print("Too high! Try lower.");
        }
        
        game.printStatus();
    }
    
    print("");
    if (game.gameWon == 1) {
        print("YOU WON in " + game.guessCount + " guesses!");
    } else {
        print("GAME OVER! The number was " + game.secretNumber);
    }
}

// Simulate an optimal strategy game
func simulateBinarySearch() -> void {
    printDoubleLine();
    print("         BINARY SEARCH STRATEGY");
    printDoubleLine();
    print("");
    
    var game:GuessGame;
    game.init(73, 10);
    
    print("Using binary search to find the number!");
    print("");
    
    var low:int = 1;
    var high:int = 100;
    
    for (var i:int = 0; i < 10; i++) {
        if (game.isGameOver() == 1) {
            break;
        }
        
        var guess:int = (low + high) / 2;
        print("Guess #" + (i + 1) + ": " + guess + " (searching " + low + "-" + high + ")");
        
        var result:int = game.makeGuess(guess);
        
        if (result == 0) {
            print("FOUND IT!");
            break;
        } else if (result < 0) {
            print("Too low, adjusting range...");
            low = guess + 1;
        } else {
            print("Too high, adjusting range...");
            high = guess - 1;
        }
        print("");
    }
    
    print("Binary search found the number in " + game.guessCount + " guesses!");
}

func main() -> void {
    simulateGame();
    
    print("");
    print("");
    
    simulateBinarySearch();
    
    print("");
    printDoubleLine();
    print("NUMBER GUESSING GAME DEMONSTRATION COMPLETE");
    printDoubleLine();
}
