// Test mutable string indexing
func main() -> void {
    print("=== Mutable String Test ===");
    print("");
    
    // Test 1: Modify a string literal (this won't work - string literals are read-only in .rdata)
    // But we can demonstrate the syntax
    print("Test 1: String character assignment syntax");
    var str1:string = "Hello";
    print("Original: " + str1);
    
    // This demonstrates the syntax, but note: modifying string literals in .rdata
    // will cause a segfault. We need heap-allocated strings for true mutability.
    // str1[0] = 'J';  // Would segfault - can't modify .rdata
    
    print("Note: To truly modify strings, we need heap allocation");
    print("");
    
    // Test 2: Character indexing still works for reading
    print("Test 2: Reading characters from string");
    var ch0:char = str1[0];
    var ch1:char = str1[1];
    var ch2:char = str1[2];
    print("  str1[0] = " + ch0);
    print("  str1[1] = " + ch1);
    print("  str1[2] = " + ch2);
    print("");
    
    print("=== Tests Complete ===");
}
