// Minimal test for array in struct methods
struct Board {
    int cells[9];
    
    func setCell(index:int, value:int) -> void {
        cells[index] = value;
    }
    
    func getCell(index:int) -> int {
        return cells[index];
    }
}

func main() -> void {
    var board:Board;
    board.setCell(0, 42);
    var result:int;
    result = board.getCell(0);
}
