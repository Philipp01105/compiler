// =================================================================
// CONVERSION UTILITIES
// =================================================================

// Converts an integer to its absolute value
func abs(n:int) -> int {
    if (n < 0) {
        return -n;
    }
    return n;
}

// Swaps two integer values (requires passing variables by reference - simulated with return)
// Helper: returns first value to swap - caller must manually swap
func swap(a:int, b:int) -> int {
    return b;
}

// =================================================================
// RANGE AND BOUNDARY UTILITIES
// =================================================================

// Clamps a value between min and max
func clamp(value:int, min:int, max:int) -> int {
    if (value < min) {
        return min;
    }
    if (value > max) {
        return max;
    }
    return value;
}

// Checks if value is within range [min, max] inclusive
func inRange(value:int, min:int, max:int) -> bit {
    if (value >= min && value <= max) {
        return 1;
    }
    return 0;
}

// Checks if value is within range (min, max) exclusive
func inRangeExclusive(value:int, min:int, max:int) -> bit {
    if (value > min && value < max) {
        return 1;
    }
    return 0;
}

// =================================================================
// COMPARISON UTILITIES
// =================================================================

// Returns the minimum of two integers
func min(a:int, b:int) -> int {
    if (a < b) {
        return a;
    }
    return b;
}

// Returns the maximum of two integers
func max(a:int, b:int) -> int {
    if (a > b) {
        return a;
    }
    return b;
}

// Returns the minimum of three integers
func min3(a:int, b:int, c:int) -> int {
    var temp:int = min(a, b);
    return min(temp, c);
}

// Returns the maximum of three integers
func max3(a:int, b:int, c:int) -> int {
    var temp:int = max(a, b);
    return max(temp, c);
}

// Returns the minimum of four integers
func min4(a:int, b:int, c:int, d:int) -> int {
    var temp1:int = min(a, b);
    var temp2:int = min(c, d);
    return min(temp1, temp2);
}

// Returns the maximum of four integers
func max4(a:int, b:int, c:int, d:int) -> int {
    var temp1:int = max(a, b);
    var temp2:int = max(c, d);
    return max(temp1, temp2);
}

// =================================================================
// SIGN AND POLARITY
// =================================================================

// Returns the sign of a number: -1, 0, or 1
func sign(n:int) -> int {
    if (n < 0) {
        return -1;
    }
    if (n > 0) {
        return 1;
    }
    return 0;
}

// Returns 1 if numbers have same sign, 0 otherwise
func sameSign(a:int, b:int) -> bit {
    if ((a >= 0 && b >= 0) || (a < 0 && b < 0)) {
        return 1;
    }
    return 0;
}

// =================================================================
// CONDITIONAL UTILITIES
// =================================================================

// Returns a if condition is true, otherwise returns b
func selectInt(condition:bit, a:int, b:int) -> int {
    if (condition == 1) {
        return a;
    }
    return b;
}

// Returns 1 if all three bits are true
func all3(a:bit, b:bit, c:bit) -> bit {
    if (a == 1 && b == 1 && c == 1) {
        return 1;
    }
    return 0;
}

// Returns 1 if any of three bits is true
func any3(a:bit, b:bit, c:bit) -> bit {
    if (a == 1 || b == 1 || c == 1) {
        return 1;
    }
    return 0;
}

// Returns 1 if exactly one bit is true
func exactlyOne(a:bit, b:bit) -> bit {
    if ((a == 1 && b == 0) || (a == 0 && b == 1)) {
        return 1;
    }
    return 0;
}

// =================================================================
// PRINTING UTILITIES
// =================================================================

// Prints a separator line of given character
func printSeparator(c:char, length:int) -> void {
    for (var i:int = 0; i < length; i++) {
        print("" + c);
    }
}

// Prints a line of dashes
func printLine() -> void {
    printSeparator('-', 50);
}

// Prints a line of equals signs
func printDoubleLine() -> void {
    printSeparator('=', 50);
}

// Prints n newlines
func printNewlines(n:int) -> void {
    for (var i:int = 0; i < n; i++) {
        print("");
    }
}

// Prints an integer with a label
func printLabeled(label:string, value:int) -> void {
    print(label + ": " + value);
}

// Prints a boolean value as "true" or "false"
func printBool(value:bit) -> void {
    if (value == 1) {
        print("true");
    } else {
        print("false");
    }
}

// Prints a boolean with a label
func printLabeledBool(label:string, value:bit) -> void {
    print(label + ": " + value);
}

// =================================================================
// VALIDATION UTILITIES
// =================================================================

// Checks if a value is a valid index for an array of given size
func isValidIndex(index:int, size:int) -> bit {
    if (index >= 0 && index < size) {
        return 1;
    }
    return 0;
}

// Checks if a range [start, end) is valid for an array
func isValidRange(start:int, end:int, size:int) -> bit {
    if (start >= 0 && end <= size && start <= end) {
        return 1;
    }
    return 0;
}

// =================================================================
// BIT MANIPULATION HELPERS
// =================================================================

// Returns 1 if the bit at position pos is set in value
func isBitSet(value:int, pos:int) -> bit {
    var mask:int = 1;
    for (var i:int = 0; i < pos; i++) {
        mask = mask * 2;
    }
    var result:int = value % (mask * 2);
    if (result >= mask) {
        return 1;
    }
    return 0;
}

// Counts the number of set bits in a value (up to 16 bits)
func countBits(value:int) -> int {
    var count:int = 0;
    for (var i:int = 0; i < 16; i++) {
        if (isBitSet(value, i) == 1) {
            count = count + 1;
        }
    }
    return count;
}

// =================================================================
// ERROR CHECKING
// =================================================================

// Checks if division is safe (divisor is not zero)
func isSafeDiv(divisor:int) -> bit {
    if (divisor != 0) {
        return 1;
    }
    return 0;
}

// Returns quotient if safe, otherwise returns 0
func safeDivide(dividend:int, divisor:int) -> int {
    if (divisor == 0) {
        return 0;
    }
    return dividend / divisor;
}

// Returns remainder if safe, otherwise returns 0
func safeModulo(dividend:int, divisor:int) -> int {
    if (divisor == 0) {
        return 0;
    }
    return dividend % divisor;
}
