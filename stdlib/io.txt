// Write newline to stdout
func io_write_newline() -> int {
    var newline:char = 10;  // '\n'

}

// Print string (no newline)

}

}

// =================================================================
// File I/O Functions
// =================================================================

// Open file for reading
// Returns: file descriptor or -1 on error
func io_file_open_read(path:string) -> int {
    return sys_open(path, 0, 0);  // O_RDONLY
}

// Open file for writing (creates if doesn't exist, truncates if exists)
// Returns: file descriptor or -1 on error
func io_file_open_write(path:string) -> int {
    return sys_open(path, 577, 420);  // O_WRONLY|O_CREAT|O_TRUNC, 0644
}

// Open file for appending
// Returns: file descriptor or -1 on error
func io_file_open_append(path:string) -> int {
    return sys_open(path, 1089, 420);  // O_WRONLY|O_CREAT|O_APPEND, 0644
}

// Close file
// Returns: 0 on success, -1 on error
func io_file_close(fd:int) -> int {
    return sys_close(fd);
}

// Read from file
// Returns: number of bytes read
func io_file_read(fd:int, buffer:string, count:int) -> int {
    return sys_read(fd, buffer, count);
}

// Write to file
// Returns: number of bytes written
func io_file_write(fd:int, data:string, count:int) -> int {
    return sys_write(fd, data, count);
}

// Write string to file (calculates length automatically)
func io_file_write_str(fd:int, s:string) -> int {
    var len:int = io_strlen(s);
    return sys_write(fd, s, len);
}
