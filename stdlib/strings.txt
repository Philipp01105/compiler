// Returns the length of a string by iterating until null terminator
func strLength(s:string) -> int {
    var length:int = 0;
    for (var i:int = 0; i < 1000; i++) {
        var ch:char = s[i];
        if (ch == '\0') {
            return length;
        }
        length = length + 1;
    }
    return length;
}

// Compares two strings character by character
// Returns 0 if equal, <0 if s1 < s2, >0 if s1 > s2
func strCompare(s1:string, s2:string) -> int {
    for (var i:int = 0; i < 1000; i++) {
        var ch1:char = s1[i];
        var ch2:char = s2[i];
        
        if (ch1 != ch2) {
            var diff:int = ch1 - ch2;
            return diff;
        }
        
        if (ch1 == '\0') {
            return 0;
        }
    }
    return 0;
}

// Checks if string starts with a given character
func strStartsWith(s:string, c:char) -> bit {
    if (s[0] == c) {
        return 1;
    }
    return 0;
}

// Checks if string ends with a given character
func strEndsWith(s:string, c:char) -> bit {
    var len:int = strLength(s);
    if (len > 0) {
        var lastChar:char = s[len - 1];
        if (lastChar == c) {
            return 1;
        }
    }
    return 0;
}

// Returns the character at a specific index
// Returns '\0' if index is out of bounds
func strCharAt(s:string, index:int) -> char {
    if (index < 0) {
        return '\0';
    }
    
    var len:int = strLength(s);
    if (index >= len) {
        return '\0';
    }
    
    return s[index];
}

// Finds the first occurrence of a character in a string
// Returns -1 if not found
func strIndexOf(s:string, c:char) -> int {
    for (var i:int = 0; i < 1000; i++) {
        var ch:char = s[i];
        if (ch == '\0') {
            return -1;
        }
        if (ch == c) {
            return i;
        }
    }
    return -1;
}

// Finds the last occurrence of a character in a string
// Returns -1 if not found
func strLastIndexOf(s:string, c:char) -> int {
    var lastIndex:int = -1;
    for (var i:int = 0; i < 1000; i++) {
        var ch:char = s[i];
        if (ch == '\0') {
            return lastIndex;
        }
        if (ch == c) {
            lastIndex = i;
        }
    }
    return lastIndex;
}

// Counts occurrences of a character in a string
func strCountChar(s:string, c:char) -> int {
    var count:int = 0;
    for (var i:int = 0; i < 1000; i++) {
        var ch:char = s[i];
        if (ch == '\0') {
            return count;
        }
        if (ch == c) {
            count = count + 1;
        }
    }
    return count;
}

// =================================================================
// STRING CLASSIFICATION
// =================================================================

// Checks if all characters in string are letters
func strIsAlpha(s:string) -> bit {
    if (s[0] == '\0') {
        return 0;
    }
    
    for (var i:int = 0; i < 1000; i++) {
        var ch:char = s[i];
        if (ch == '\0') {
            return 1;
        }
        if (!((ch >= 'A' && ch <= 'Z') || (ch >= 'a' && ch <= 'z'))) {
            return 0;
        }
    }
    return 1;
}

// Checks if all characters in string are digits
func strIsDigit(s:string) -> bit {
    if (s[0] == '\0') {
        return 0;
    }
    
    for (var i:int = 0; i < 1000; i++) {
        var ch:char = s[i];
        if (ch == '\0') {
            return 1;
        }
        if (!(ch >= '0' && ch <= '9')) {
            return 0;
        }
    }
    return 1;
}

// Checks if all characters in string are alphanumeric
func strIsAlnum(s:string) -> bit {
    if (s[0] == '\0') {
        return 0;
    }
    
    for (var i:int = 0; i < 1000; i++) {
        var ch:char = s[i];
        if (ch == '\0') {
            return 1;
        }
        if (!((ch >= 'A' && ch <= 'Z') || (ch >= 'a' && ch <= 'z') || (ch >= '0' && ch <= '9'))) {
            return 0;
        }
    }
    return 1;
}

// Checks if all characters in string are uppercase
func strIsUpper(s:string) -> bit {
    if (s[0] == '\0') {
        return 0;
    }
    
    var hasLetter:bit = 0;
    for (var i:int = 0; i < 1000; i++) {
        var ch:char = s[i];
        if (ch == '\0') {
            return hasLetter;
        }
        if (ch >= 'A' && ch <= 'Z') {
            hasLetter = 1;
        } else if (ch >= 'a' && ch <= 'z') {
            return 0;
        }
    }
    return hasLetter;
}

// Checks if all characters in string are lowercase
func strIsLower(s:string) -> bit {
    if (s[0] == '\0') {
        return 0;
    }
    
    var hasLetter:bit = 0;
    for (var i:int = 0; i < 1000; i++) {
        var ch:char = s[i];
        if (ch == '\0') {
            return hasLetter;
        }
        if (ch >= 'a' && ch <= 'z') {
            hasLetter = 1;
        } else if (ch >= 'A' && ch <= 'Z') {
            return 0;
        }
    }
    return hasLetter;
}

// =================================================================
// STRING UTILITIES
// =================================================================

// Prints a string with a newline
func strPrintln(s:string) -> void {
    print(s);
}

// Prints a character n times
func strRepeatChar(c:char, n:int) -> void {
    for (var i:int = 0; i < n; i++) {
        print("" + c);
    }
}

// Counts the number of words in a string (space-separated)
func strCountWords(s:string) -> int {
    var count:int = 0;
    var inWord:bit = 0;
    
    for (var i:int = 0; i < 1000; i++) {
        var ch:char = s[i];
        if (ch == '\0') {
            return count;
        }
        
        if (ch == ' ' || ch == '\t' || ch == '\n') {
            inWord = 0;
        } else {
            if (inWord == 0) {
                count = count + 1;
                inWord = 1;
            }
        }
    }
    return count;
}

// Counts the number of lines in a string
func strCountLines(s:string) -> int {
    var count:int = 1;
    for (var i:int = 0; i < 1000; i++) {
        var ch:char = s[i];
        if (ch == '\0') {
            return count;
        }
        if (ch == '\n') {
            count = count + 1;
        }
    }
    return count;
}

// =================================================================
// STRING TRANSFORMATION HELPERS
// =================================================================

// Converts a digit character to its integer value
// Returns -1 for non-digit characters
func strCharToDigit(c:char) -> int {
    if (c >= '0' && c <= '9') {
        return c - '0';
    }
    return -1;
}

// Converts an integer digit (0-9) to its character representation
func strDigitToChar(digit:int) -> char {
    if (digit >= 0 && digit <= 9) {
        var ch:char = '0' + digit;
        return ch;
    }
    return '?';
}

// Parses a positive integer from a string
// Returns -1 if string doesn't represent a valid number
func strParseInt(s:string) -> int {
    var result:int = 0;
    var len:int = strLength(s);
    
    if (len == 0) {
        return -1;
    }
    
    for (var i:int = 0; i < len; i++) {
        var ch:char = s[i];
        var digit:int = strCharToDigit(ch);
        
        if (digit == -1) {
            return -1;
        }
        
        result = result * 10;
        result = result + digit;
    }
    
    return result;
}
