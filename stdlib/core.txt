// =================================================================
// INTEGER OPERATIONS
// =================================================================

// Returns the absolute value of an integer
func intAbs(n:int) -> int {
    if (n < 0) {
        return -n;
    }
    return n;
}

// Returns the minimum of two integers
func intMin(a:int, b:int) -> int {
    if (a < b) {
        return a;
    }
    return b;
}

// Returns the maximum of two integers
func intMax(a:int, b:int) -> int {
    if (a > b) {
        return a;
    }
    return b;
}

// Clamps a value between min and max
func intClamp(value:int, min:int, max:int) -> int {
    if (value < min) {
        return min;
    }
    if (value > max) {
        return max;
    }
    return value;
}

// Checks if an integer is even
func intIsEven(n:int) -> bit {
    if (n % 2 == 0) {
        return 1;
    }
    return 0;
}

// Checks if an integer is odd
func intIsOdd(n:int) -> bit {
    if (n % 2 != 0) {
        return 1;
    }
    return 0;
}

// Checks if an integer is positive
func intIsPositive(n:int) -> bit {
    if (n > 0) {
        return 1;
    }
    return 0;
}

// Checks if an integer is negative
func intIsNegative(n:int) -> bit {
    if (n < 0) {
        return 1;
    }
    return 0;
}

// Checks if an integer is zero
func intIsZero(n:int) -> bit {
    if (n == 0) {
        return 1;
    }
    return 0;
}

// Returns the sign of an integer (-1, 0, or 1)
func intSign(n:int) -> int {
    if (n < 0) {
        return -1;
    }
    if (n > 0) {
        return 1;
    }
    return 0;
}

// =================================================================
// STRING OPERATIONS
// =================================================================

// Checks if two strings are equal
func strEquals(a:string, b:string) -> bit {
    if (a == b) {
        return 1;
    }
    return 0;
}

// Checks if a string is empty
func strIsEmpty(s:string) -> bit {
    if (s == "") {
        return 1;
    }
    return 0;
}

// Checks if a string is not empty
func strIsNotEmpty(s:string) -> bit {
    if (s != "") {
        return 1;
    }
    return 0;
}

// =================================================================
// BOOLEAN OPERATIONS
// =================================================================

// Logical AND
func boolAnd(a:bit, b:bit) -> bit {
    if (a == 1 && b == 1) {
        return 1;
    }
    return 0;
}

// Logical OR
func boolOr(a:bit, b:bit) -> bit {
    if (a == 1 || b == 1) {
        return 1;
    }
    return 0;
}

// Logical NOT
func boolNot(b:bit) -> bit {
    if (b == 1) {
        return 0;
    }
    return 1;
}

// Logical XOR (exclusive OR)
func boolXor(a:bit, b:bit) -> bit {
    if ((a == 1 && b == 0) || (a == 0 && b == 1)) {
        return 1;
    }
    return 0;
}

// Logical NAND (NOT AND)
func boolNand(a:bit, b:bit) -> bit {
    if (a == 1 && b == 1) {
        return 0;
    }
    return 1;
}

// Logical NOR (NOT OR)
func boolNor(a:bit, b:bit) -> bit {
    if (a == 0 && b == 0) {
        return 1;
    }
    return 0;
}

// =================================================================
// BYTE OPERATIONS
// =================================================================

// Adds two bytes
func byteAdd(a:byte, b:byte) -> byte {
    var result:byte = a + b;
    return result;
}

// Subtracts two bytes
func byteSub(a:byte, b:byte) -> byte {
    var result:byte = a - b;
    return result;
}

// Returns minimum of two bytes
func byteMin(a:byte, b:byte) -> byte {
    if (a < b) {
        return a;
    }
    return b;
}

// Returns maximum of two bytes
func byteMax(a:byte, b:byte) -> byte {
    if (a > b) {
        return a;
    }
    return b;
}

// =================================================================
// CHARACTER OPERATIONS
// =================================================================

// Checks if a character is uppercase letter
func charIsUpper(c:char) -> bit {
    if (c >= 'A' && c <= 'Z') {
        return 1;
    }
    return 0;
}

// Checks if a character is lowercase letter
func charIsLower(c:char) -> bit {
    if (c >= 'a' && c <= 'z') {
        return 1;
    }
    return 0;
}

// Checks if a character is a letter
func charIsAlpha(c:char) -> bit {
    if ((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z')) {
        return 1;
    }
    return 0;
}

// Checks if a character is a digit
func charIsDigit(c:char) -> bit {
    if (c >= '0' && c <= '9') {
        return 1;
    }
    return 0;
}

// Checks if a character is alphanumeric
func charIsAlnum(c:char) -> bit {
    if ((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9')) {
        return 1;
    }
    return 0;
}

// Checks if a character is whitespace
func charIsSpace(c:char) -> bit {
    if (c == ' ' || c == '\t' || c == '\n' || c == '\r') {
        return 1;
    }
    return 0;
}

// =================================================================
// UNIVERSAL OPERATIONS (toString equivalents via print)
// =================================================================
// Note: Since we don't have dynamic string concatenation yet,
// these functions print the values with descriptive labels

func intToString(value:int) -> void {
    print("" + value);
}

func charToString(value:char) -> void {
    print("" + value);
}

func byteToString(value:byte) -> void {
    print("" + value);
}

func bitToString(value:bit) -> void {
    if (value == 1) {
        print("true");
    } else {
        print("false");
    }
}

func stringToString(value:string) -> void {
    print(value);
}
